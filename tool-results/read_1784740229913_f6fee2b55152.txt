   176→   176→            {landingLinks.map((link) => (
   177→   177→              <a key={link.href} href={link.href} className="nav-link">
   178→   178→                {link.label}

   180→   180→            ))}

   197→   197→                onClick={onOpenDashboard}
   198→   198→                variant="ghost"

   201→   201→                Dashboard

   204→   204→                onClick={onOpenDashboard}

   215→   215→          </button>
   216→   216→        </div>
   217→   217→      </div>
   218→   218→
   219→   219→      {mobileOpen && (
   220→   220→        <motion.div
   221→   221→          initial={{ opacity: 0, height: 0 }}
   222→   222→          animate={{ opacity: 1, height: "auto" }}
   223→   223→          className="lg:hidden pb-4 border-t border-white/5 px-5"
   224→   224→        >
   225→   225→          {!isDashboard &&
   226→   226→            landingLinks.map((link) => (
   227→   227→              <a
   228→   228→                key={link.href}
   229→   229→                href={link.href}
   230→   230→                onClick={() => setMobileOpen(false)}
   231→   231→                className="block py-3 text-xs uppercase tracking-wider text-[#888899]"
   232→   232→              >
   233→   233→                {link.label}

   239→   239→            style={{ color: "#FF4D00" }}
   240→   240→          >
   241→   241→            {isDashboard ? "Back to Site" : "Launch Dashboard"}
   242→   242→          </button>
   243→   243→        </motion.div>
   244→   244→      )}
   245→   245→    </motion.nav>
   246→   246→  );
   247→   247→}
